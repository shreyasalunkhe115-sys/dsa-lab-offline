StackArrayDemo.java
import java.util.Scanner;

class StackArray {

    int stack[];

    int top;

    int size;

    StackArray(int size) {

        this.size = size;

        stack = new int[size];

        top = -1;

    }

    // Push operation

    void push(int value) {

        if (top == size - 1) {

            System.out.println("Stack Overflow");

        } else {

            stack[++top] = value;

            System.out.println(value + " inserted into stack.");

        }

    }

    // Pop operation

    void pop() {

        if (top == -1) {

            System.out.println("Stack Underflow");

        } else {

            System.out.println("Deleted Element: " + stack[top--]);

        }

    }

    // Display operation

    void display() {

        if (top == -1) {

            System.out.println("Stack is Empty");

        } else {

            System.out.println("Stack Elements:");

            for (int i = top; i >= 0; i--) {

                System.out.println(stack[i]);

            }        }

    }  }

        if (top == null) {

import java.util.Scanner;

class QueueADT {
    static final int SIZE = 5;
    int[] queue = new int[SIZE];
    int front = -1, rear = -1;

    // Linear Queue Enqueue
    void linearEnqueue(int data) {
        if (rear == SIZE - 1) {
            System.out.println("Linear Queue Overflow");
        } else {
            if (front == -1)
                front = 0;
            queue[++rear] = data;
            System.out.println(data + " inserted.");
        }
    }

    // Linear Queue Dequeue
    void linearDequeue() {
        if (front == -1 || front > rear) {
            System.out.println("Linear Queue Underflow");
        } else {
            System.out.println("Deleted: " + queue[front++]);
        }
    }

    // Linear Queue Display
    void linearDisplay() {
        if (front == -1 || front > rear) {
            System.out.println("Queue is Empty");
        } else {
            System.out.print("Linear Queue: ");
            for (int i = front; i <= rear; i++)
                System.out.print(queue[i] + " ");
            System.out.println();
        }
    }
}

class CircularQueue {
    static final int SIZE = 5;
    int[] queue = new int[SIZE];
    int front = -1, rear = -1;

    // Circular Queue Enqueue
    void enqueue(int data) {
        if ((rear + 1) % SIZE == front) {
            System.out.println("Circular Queue Overflow");
        } else {
            if (front == -1)
                front = 0;
            rear = (rear + 1) % SIZE;
            queue[rear] = data;
            System.out.println(data + " inserted.");
        }
    }

    // Circular Queue Dequeue
    void dequeue() {
        if (front == -1) {
            System.out.println("Circular Queue Underflow");
        } else {
            System.out.println("Deleted: " + queue[front]);

            if (front == rear) {
                front = rear = -1;
            } else {
                front = (front + 1) % SIZE;
            }
        }
    }

    // Circular Queue Display
    void display() {
        if (front == -1) {
            System.out.println("Queue is Empty");
            return;
        }

        System.out.print("Circular Queue: ");
        int i = front;
        while (true) {
            System.out.print(queue[i] + " ");
            if (i == rear)
                break;
            i = (i + 1) % SIZE;
        }
        System.out.println();
    }
}

3Queue application: Round Robin scheduling

CPU scheduling with a time quantum.

Fixed: your original had no main method, so java could not start it. I added main and used a real Queue. The results match your sample output.

RoundRobinScheduling.java
import java.util.*;

public class RoundRobinScheduling {

    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        System.out.print("Enter Number of Processes: ");
        int n = sc.nextInt();

        int[] burstTime = new int[n];
        int[] remainingTime = new int[n];
        int[] waitingTime = new int[n];
        int[] turnaroundTime = new int[n];

        System.out.println("Enter Burst Time of Processes:");
        for (int i = 0; i < n; i++) {
            System.out.print("P" + (i + 1) + ": ");
            burstTime[i] = sc.nextInt();
            remainingTime[i] = burstTime[i];
        }

        System.out.print("Enter Time Quantum: ");
        int quantum = sc.nextInt();

        // Ready queue holds process indices (FIFO)
        Queue<Integer> queue = new LinkedList<>();
        for (int i = 0; i < n; i++)
            queue.add(i);

        int time = 0;

        while (!queue.isEmpty()) {
            int i = queue.poll();

            if (remainingTime[i] > quantum) {
                time += quantum;
                remainingTime[i] -= quantum;
                queue.add(i);                 // back to end of queue
            } else {
                time += remainingTime[i];
                remainingTime[i] = 0;
                turnaroundTime[i] = time;     // all arrive at time 0
                waitingTime[i] = time - burstTime[i];
            }
        }

        double totalWT = 0, totalTAT = 0;

        System.out.println("\nProcess\tBurst\tWaiting\tTurnaround");
        for (int i = 0; i < n; i++) {
            totalWT += waitingTime[i];
            totalTAT += turnaroundTime[i];
            System.out.println("P" + (i + 1) + "\t" + burstTime[i] + "\t"
                    + waitingTime[i] + "\t" + turnaroundTime[i]);
        }

        System.out.printf("\nAverage Waiting Time = %.2f", totalWT / n);
        System.out.printf("\nAverage Turnaround Time = %.2f\n", totalTAT / n);

        sc.close();
    }
}

Hash table with collision resolution by chaining.

HashingChaining.java
import java.util.Scanner;

class Node {

    int key;

    Node next;

    Node(int key) {

        this.key = key;

        next = null;

    }

}

class HashTable {

    private Node[] table;

    private int size;

    HashTable(int size) {

        this.size = size;

        table = new Node[size];

    }

    private int hashFunction(int key) {

        return key % size;

    }

    // Insert

    public void insert(int key) {

        int index = hashFunction(key);

        Node newNode = new Node(key);

        if (table[index] == null) {

            table[index] = newNode;

        } else {

            Node temp = table[index];

            while (temp.next != null) {

                temp = temp.next;

            }

            temp.next = newNode;

        }

        System.out.println(key + " inserted successfully.");

    }

    // Search

    public void search(int key) {

        int index = hashFunction(key);

        Node temp = table[index];

        while (temp != null) {

            if (temp.key == key) {

                System.out.println(key + " found.");

                return;

            }

            temp = temp.next;

        }

        System.out.println(key + " not found.");

    }

    // Delete

    public void delete(int key) {

        int index = hashFunction(key);

        Node temp = table[index];

        Node prev = null;

        while (temp != null && temp.key != key) {

            prev = temp;

            temp = temp.next;

        }

        if (temp == null) {

            System.out.println("Key not found.");

            return;

        }

        if (prev == null) {

            table[index] = temp.next;

        } else {

            prev.next = temp.next;

        }

        System.out.println(key + " deleted successfully.");

    }

    // Display

    public void display() {

        System.out.println("\nHash Table:");

        for (int i = 0; i < size; i++) {

            System.out.print(i + " --> ");

            Node temp = table[i];

            while (temp != null) {

                System.out.print(temp.key + " -> ");

                temp = temp.next;

            }

            System.out.println("NULL");

        }

    }

}

5Binary Search Tree

Insert, delete, search, traversals, level order and height.

Exp05_BST.java
import java.util.*;

/** Experiment 5: Binary Search Tree (BST) Abstract Data Type in Java. */
public class Exp05_BST {

    static class Node {
        int data; Node left, right;
        Node(int d) { data = d; }
    }

    private Node root;

    Node insert(Node node, int key) {
        if (node == null) return new Node(key);
        if (key < node.data) node.left = insert(node.left, key);
        else if (key > node.data) node.right = insert(node.right, key);
        else System.out.println("Duplicate key ignored.");
        return node;
    }

    boolean search(Node node, int key) {
        if (node == null) return false;
        if (key == node.data) return true;
        return key < node.data ? search(node.left, key) : search(node.right, key);
    }

    Node minNode(Node n) { while (n.left != null) n = n.left; return n; }

    Node delete(Node node, int key) {
        if (node == null) { System.out.println("Key not found."); return null; }
        if (key < node.data) node.left = delete(node.left, key);
        else if (key > node.data) node.right = delete(node.right, key);
        else {
            if (node.left == null) return node.right;      // 0 or 1 child
            if (node.right == null) return node.left;
            Node succ = minNode(node.right);                // 2 children
            node.data = succ.data;
            node.right = delete(node.right, succ.data);
        }
        return node;
    }

    void inorder(Node n)   { if (n != null) { inorder(n.left); System.out.print(n.data + " "); inorder(n.right); } }
    void preorder(Node n)  { if (n != null) { System.out.print(n.data + " "); preorder(n.left); preorder(n.right); } }
    void postorder(Node n) { if (n != null) { postorder(n.left); postorder(n.right); System.out.print(n.data + " "); } }

    int height(Node n) { return n == null ? 0 : 1 + Math.max(height(n.left), height(n.right)); }

    void levelOrder() {
        if (root == null) { System.out.println("Tree is empty."); return; }
        Queue<Node> q = new LinkedList<>();
        q.add(root);
        while (!q.isEmpty()) {
            Node n = q.poll();
            System.out.print(n.data + " ");
            if (n.left != null) q.add(n.left);
            if (n.right != null) q.add(n.right);
        }
        System.out.println();
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        Exp05_BST t = new Exp05_BST();
        int ch;
        do {
            System.out.println("\n--- BST MENU ---");
            System.out.println("1.Insert 2.Delete 3.Search 4.Inorder 5.Preorder 6.Postorder 7.Level Order 8.Height 0.Exit");
            System.out.print("Choice: ");
            ch = sc.nextInt();
            switch (ch) {
                case 1: System.out.print("Value: "); t.root = t.insert(t.root, sc.nextInt()); break;
                case 2: System.out.print("Value: "); t.root = t.delete(t.root, sc.nextInt()); break;
                case 3: System.out.print("Value: "); System.out.println(t.search(t.root, sc.nextInt()) ? "Found" : "Not found"); break;
                case 4: System.out.print("Inorder: ");   t.inorder(t.root);   System.out.println(); break;
                case 5: System.out.print("Preorder: ");  t.preorder(t.root);  System.out.println(); break;
                case 6: System.out.print("Postorder: "); t.postorder(t.root); System.out.println(); break;
                case 7: System.out.print("Level order: "); t.levelOrder(); break;
                case 8: System.out.println("Height = " + t.height(t.root)); break;
                case 0: System.out.println("Exiting..."); break;
                default: System.out.println("Invalid choice!");
            }
        } while (ch != 0);
        sc.close();
    }
}

6BFS traversal

Breadth-first search using an adjacency matrix and an adjacency list.

The graph is built into the program, so there is no input.

BFSMatrix.java
import java.util.*;

public class BFSMatrix {

    private int vertices;

    private int[][] matrix;

    public BFSMatrix(int vertices) {

        this.vertices = vertices;

        matrix = new int[vertices][vertices];

    }

    public void addEdge(int source, int destination) {

        matrix[source][destination] = 1;

        matrix[destination][source] = 1;

    }

    public void bfs(int start) {

        boolean[] visited = new boolean[vertices];

        Queue<Integer> queue = new LinkedList<>();

        visited[start] = true;

        queue.add(start);

        System.out.print("BFS Traversal : ");

        while (!queue.isEmpty()) {

            int node = queue.poll();

            System.out.print(node + " ");

            for (int i = 0; i < vertices; i++) {

                if (matrix[node][i] == 1 && !visited[i]) {

                    visited[i] = true;

                    queue.add(i);

                }

            }

        }

    }

    public static void main(String args[]) {

        BFSMatrix graph = new BFSMatrix(6);

        graph.addEdge(0,1);

        graph.addEdge(0,2);

        graph.addEdge(1,3);

        graph.addEdge(1,4);

        graph.addEdge(2,5);

        graph.addEdge(4,5);

        graph.bfs(0);

    }

}

9Binary search

Search a sorted array by halving the range.

The array must be entered in sorted order.

BinarySearchDemo.java
import java.util.Scanner;

public class BinarySearchDemo {

    public static int binarySearch(int arr[], int key) {

        int low = 0;

        int high = arr.length - 1;

        while (low <= high) {

            int mid = (low + high) / 2;

            if (arr[mid] == key)

                return mid;

            else if (key < arr[mid])

                high = mid - 1;

            else

                low = mid + 1;

        }

        return -1;

    }

    public static void main(String args[]) {

        Scanner sc = new Scanner(System.in);

        System.out.print("Enter number of elements: ");

        int n = sc.nextInt();

        int arr[] = new int[n];

        System.out.println("Enter sorted array elements:");

        for (int i = 0; i < n; i++) {

            arr[i] = sc.nextInt();

        }

        System.out.print("Enter element to search: ");

        int key = sc.nextInt();

        int result = binarySearch(arr, key);

        if (result != -1)

            System.out.println("Element found at index: " + result);

        else

            System.out.println("Element not found.");

        sc.close();

    }

10Bubble and selection sort

Two simple O(n²) sorts.

BubbleSort.java
import java.util.Scanner;

public class BubbleSort {

    public static void bubbleSort(int arr[]) {

        int n = arr.length;

        for (int i = 0; i < n - 1; i++) {

            boolean swapped = false;

            for (int j = 0; j < n - i - 1; j++) {

                if (arr[j] > arr[j + 1]) {

                    int temp = arr[j];

                    arr[j] = arr[j + 1];

                    arr[j + 1] = temp;

                    swapped = true;

                }

            }

            if (!swapped)

                break;

        }

    }

    public static void display(int arr[]) {

        for (int num : arr)

            System.out.print(num + " ");

        System.out.println();

    }

    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        System.out.print("Enter number of elements: ");

        int n = sc.nextInt();

        int arr[] = new int[n];

        System.out.println("Enter Elements:");

        for (int i = 0; i < n; i++)

            arr[i] = sc.nextInt();

        System.out.println("\nOriginal Array:");

        display(arr);

        bubbleSort(arr);

        System.out.println("Sorted Array:");

        display(arr);

        sc.close();

    }

}

    public static void selectionSort(int arr[]) {

        int n = arr.length;

11Merge and quick sort

Two divide-and-conquer sorts.

MergeSort.java
import java.util.Scanner;

public class MergeSort {

    void merge(int arr[], int l, int m, int r) {

        int n1 = m - l + 1;

        int n2 = r - m;

        int L[] = new int[n1];

        int R[] = new int[n2];

        for (int i = 0; i < n1; i++)

            L[i] = arr[l + i];

        for (int j = 0; j < n2; j++)

            R[j] = arr[m + 1 + j];

        int i = 0, j = 0;

        int k = l;

        while (i < n1 && j < n2) {

            if (L[i] <= R[j]) {

                arr[k] = L[i];

                i++;

            }

            else {

                arr[k] = R[j];

                j++;

            }

            k++;

        }

        while (i < n1) {

            arr[k] = L[i];

            i++;

            k++;

        }

        while (j < n2) {

            arr[k] = R[j];

            j++;

            k++;

        }

    }

    void sort(int arr[], int l, int r) {

        if (l < r) {

            int m = (l + r) / 2;

            sort(arr, l, m);

            sort(arr, m + 1, r);

            merge(arr, l, m, r);

        }

    }

    void display(int arr[]) {

        for (int i : arr)

            System.out.print(i + " ");

    }

    public static void main(String args[]) {

        Scanner sc = new Scanner(System.in);

        System.out.print("Enter number of elements: ");

        int n = sc.nextInt();

        int arr[] = new int[n];

        System.out.println("Enter Elements:");

        for (int i = 0; i < n; i++)

            arr[i] = sc.nextInt();

        MergeSort ob = new MergeSort();

        ob.sort(arr, 0, n - 1);

        System.out.println("Sorted Array:");

        ob.display(arr);

    }

}

QuickSort.java
import java.util.Scanner;

public class QuickSort {

    int partition(int arr[], int low, int high) {

        int pivot = arr[high];

        int i = low - 1;

        for (int j = low; j < high; j++) {

            if (arr[j] < pivot) {

                i++;

                int temp = arr[i];

                arr[i] = arr[j];

                arr[j] = temp;

            }

        }

        int temp = arr[i + 1];

        arr[i + 1] = arr[high];

        arr[high] = temp;

        return i + 1;

    }

    void quickSort(int arr[], int low, int high) {

        if (low < high) {

            int pi = partition(arr, low, high);

            quickSort(arr, low, pi - 1);

            quickSort(arr, pi + 1, high);

        }

    }

    void display(int arr[]) {

        for (int i : arr)

            System.out.print(i + " ");

    }

    public static void main(String args[]) {

        Scanner sc = new Scanner(System.in);

        System.out.print("Enter number of elements: ");

        int n = sc.nextInt();

        int arr[] = new int[n];

        System.out.println("Enter Elements:");

        for (int i = 0; i < n; i++)

            arr[i] = sc.nextInt();

        QuickSort obj = new QuickSort();

        obj.quickSort(arr, 0, n - 1);

        System.out.println("Sorted Array:");

        obj.display(arr);

    }

}
